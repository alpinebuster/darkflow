4#[derive(Clone)]
5pub struct PacketSequence {
6    dirs: Vec<i32>,        // Direction (/ -)
7    lens: Vec<i32>,
8    timestamps: Vec<i64>,  // Relative time us
9    first_ts: Option<i64>,
10
11    // ===== TCP flags =====
12    window: Vec<u16>,
13
14    /*
15     | Flag| Binary | Function / Purpose        | Common Usage / Scenario                                                  |
16     | --- | ------ | ------------------------- | ------------------------------------------------------------------------ |
17     | FIN | 0x01   | Close connection          | Very important, indicates flow end                                       |
18     | SYN | 0x02   | Establish connection      | Very important, distinguishes new flows                                  |
19     | RST | 0x04   | Reset connection          | Important, detects errors / failed connections                           |
20     | PSH | 0x08   | Push data                 | Less common, indicates application layer data should be sent immediately |
21     | ACK | 0x10   | Acknowledgment valid      | Extremely common, present in almost every packet                         |
22     | URG | 0x20   | Urgent pointer            | Rarely used                                                              |
23     | ECE | 0x40   | ECN Echo                  | Rarely used, network congestion flag                                     |
24     | CWR | 0x80   | Congestion Window Reduced | Rarely used, usually with ECN                                            |
25     */
26    fin: Vec<u8>,
27    syn: Vec<u8>,
28    rst: Vec<u8>,
29    psh: Vec<u8>,
30    ack: Vec<u8>,
31    urg: Vec<u8>, // Rarely used
32    cwr: Vec<u8>, // Rarely used
33    ece: Vec<u8>, // Rarely used
34
35    // ===== IP =====
36    ttl: Vec<u8>,
37    ip_df: Vec<u8>,
38    ip_mf: Vec<u8>,
39}

63    fn vec_to_string<T: ToString>(v: &[T]) -> String {
64        let inner = v.iter()
65            .map(|x| x.to_string())
66            .collect::<Vec<_>>()
67            .join(",");
68        format!("\"[{}]\"", inner)

72impl FlowFeature for PacketSequence {
73    fn update(&mut self, packet: &PacketFeatures, fwd: bool, _last_ts: i64) {
74        let ts = packet.timestamp_us;
75
76        // Initialize the time of the first package
77        let first = self.first_ts.get_or_insert(ts);
78
79        self.dirs.push(if fwd { 1 } else { -1 });
80        // Packet length
81        self.lens.push(packet.length as i32);
82
83        // Relative time
84        self.timestamps.push(ts - *first);
85
86        // ===== TCP window =====
87        self.window.push(packet.window_size);
88
89        // ===== TCP flags =====
90        self.fin.push(packet.fin_flag);
91        self.syn.push(packet.syn_flag);
92        self.rst.push(packet.rst_flag);
93        self.psh.push(packet.psh_flag);
94        self.ack.push(packet.ack_flag);
95        self.urg.push(packet.urg_flag);
96        self.cwr.push(packet.cwr_flag);
97        self.ece.push(packet.ece_flag);
98
99        // ===== IP =====
100        self.ttl.push(packet.ttl);
101        self.ip_df.push(packet.ip_df);
102        self.ip_mf.push(packet.ip_mf);
103    }
104
105    fn close(&mut self, _timestamp: i64, _cause: FlowExpireCause) {}
106
107    fn dump(&self) -> String {
108        format!(
109            "{},{},{},{},{},{},{},{},{},{},{},{},{},{},{}",
110            Self::vec_to_string(&self.dirs),
111            // Pure packet length sequence
112            Self::vec_to_string(
113                &self.lens.iter().map(|v| v.abs()).collect::<Vec<_>>()
114            ),
115            // Time series sequence
116            Self::vec_to_string(&self.timestamps),
117
118            Self::vec_to_string(&self.window),
119            Self::vec_to_string(&self.fin),
120            Self::vec_to_string(&self.syn),
121            Self::vec_to_string(&self.rst),
122            Self::vec_to_string(&self.psh),
123            Self::vec_to_string(&self.ack),
124            Self::vec_to_string(&self.urg),
125            Self::vec_to_string(&self.cwr),
126            Self::vec_to_string(&self.ece),
127
128            Self::vec_to_string(&self.ttl),
129            Self::vec_to_string(&self.ip_df),
130            Self::vec_to_string(&self.ip_mf),
131        )
132    }
