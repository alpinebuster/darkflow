12/*
13Multivariate Time Series (MTS) classification traffic features.
14
15References:
16    [1] Fauvel, Kevin, Fuxing Chen, and Dario Rossi.
17    "A lightweight, efficient and explainable-by-design convolutional neural network for
18    internet traffic classification." Proceedings of the 29th ACM SIGKDD Conference on
19    Knowledge Discovery and Data Mining. 2023.
20
21Examples:
22    Imagine you have a dataset with n_samples = 3 and n_packets = 4.
23    A sample in X_train might look like this before processing:
24
25    X_train = [[10, -20, 30, -40],
26                [-5, 15, -25, 35],
27                [100, -200, 300, -400]]
28
29    Step 1: Absolute Values (X1):
30    X1 = [[10, 20, 30, 40],
31            [5, 15, 25, 35],
32            [100, 200, 300, 400]]
33
34    Step 2: Signs (X2):
35    X2 = [[1, -1, 1, -1],
36            [-1, 1, -1, 1],
37            [1, -1, 1, -1]]
38
39    Step 3: Concatenation (X3):
40    X3 = [[10, 20, 30, 40, 1, -1, 1, -1],
41            [5, 15, 25, 35, -1, 1, -1, 1],
42            [100, 200, 300, 400, 1, -1, 1, -1]]
43
44    Step 4: Reshaping:
45    X_train_reshaped = [[[[10, 1], [20, -1], [30, 1], [40, -1]]],
46                        [[[5, -1], [15, 1], [25, -1], [35, 1]]],
47                        [[[100, 1], [200, -1], [300, 1], [400, -1]]]]
48*/

50#[derive(Clone)]
51pub struct Lexnetflow {
52    /// Choose here for an existing flow type or leave the basic flow.
53    pub basic_flow: BasicFlow,
54    /// The additional features.
55    pub packet_seq: PacketSequence,
56}
57
58impl Flow for Lexnetflow {
59    fn new(
60        flow_key: String,
61        ipv4_source: IpAddr,
62        port_source: u16,
63        ipv4_destination: IpAddr,
64        port_destination: u16,
65        protocol: u8,
66        timestamp_us: i64,
67    ) -> Self {
68        Lexnetflow {
69            basic_flow: BasicFlow::new(
70                flow_key,
71                ipv4_source,
72                port_source,
73                ipv4_destination,
74                port_destination,
75                protocol,
76                timestamp_us,
77            ),
78            // The initialization of the additional features.
79            packet_seq: PacketSequence::new(),
80        }
81    }
82
83    fn update_flow(&mut self, packet: &PacketFeatures, fwd: bool) -> bool {
84        // Update the basic flow and returns true if the flow is terminated.
85        let last_timestamp_us = self.basic_flow.last_timestamp_us;
86        let is_terminated = self.basic_flow.update_flow(packet, fwd);
87
88        // The update of the additional features.
89        self.packet_seq.update(packet, fwd, last_timestamp_us);
90
91        // Return the termination status of the flow.
92        is_terminated
93    }
94
95    fn close_flow(&mut self, timestamp_us: i64, cause: FlowExpireCause) {
96        self.basic_flow.close_flow(timestamp_us, cause);
97
98        self.packet_seq.close(timestamp_us, cause);
99    }
100
101    fn dump(&self) -> String {
102        format!(
103            "{},{},{},{},{},{},{},{},{}",
104            // Basic Info
105            self.basic_flow.flow_key,
106            self.basic_flow.ip_source,
107            self.basic_flow.port_source,
108            self.basic_flow.ip_destination,
109            self.basic_flow.port_destination,
110            self.basic_flow.protocol,
111            self.basic_flow.get_first_timestamp(),
112            self.basic_flow.get_flow_duration_usec(),
113            // Lexnetflow Info
114            self.packet_seq.dump_dir_len(),
115        )
116    }
117
118    fn get_features() -> String {
119        format!(
120            "flow_key,ip_source,port_source,ip_destination,port_destination,\
121            protocol,first_timestamp,flow_duration_usec,{}",
122            PacketSequence::headers_dir_len()
123        )
124    }
125
126    fn dump_without_contamination(&self) -> String {
127        format!(
128            "{}", self.packet_seq.dump_dir_len()
129        )
130    }
131
132    fn get_features_without_contamination() -> String {
133        PacketSequence::headers_dir_len()
134    }
135
136    fn get_first_timestamp_us(&self) -> i64 {
137        self.basic_flow.first_timestamp_us
138    }
139
140    fn is_expired(
141        &self,
142        timestamp_us: i64,
143        active_timeout: u64,
144        idle_timeout: u64,
145    ) -> (bool, FlowExpireCause) {
146        self.basic_flow
147            .is_expired(timestamp_us, active_timeout, idle_timeout)
148    }
149
150    fn flow_key(&self) -> &String {
151        &self.basic_flow.flow_key
152    }
153}
